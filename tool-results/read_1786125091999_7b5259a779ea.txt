   101→// ─── Status Badge Helper ─────────────────────────────────────────
   102→
   103→const statusConfig: Record<string, { label: string; className: string }> = {
   104→  pending: { label: "En attente", className: "bg-yellow-100 text-yellow-800 dark:bg-yellow-900/30 dark:text-yellow-400" },
   105→  confirmed: { label: "Confirmé", className: "bg-blue-100 text-blue-800 dark:bg-blue-900/30 dark:text-blue-400" },
   106→  completed: { label: "Terminé", className: "bg-green-100 text-green-800 dark:bg-green-900/30 dark:text-green-400" },
   107→  cancelled: { label: "Annulé", className: "bg-red-100 text-red-800 dark:bg-red-900/30 dark:text-red-400" },
   108→};
   109→
   110→function StatusBadge({ status }: { status: string }) {
   111→  const cfg = statusConfig[status] || statusConfig.pending;
   112→  return <Badge variant="outline" className={cfg.className}>{cfg.label}</Badge>;
   113→}

   116→
   117→export default function TelegramPage() {
   118→  const { token } = useAppStore();
   119→
   120→  // Data states
   121→  const [agents, setAgents] = useState<TelegramAgent[]>([]);
   122→  const [bookings, setBookings] = useState<TelegramBooking[]>([]);
   123→  const [activeTab, setActiveTab] = useState("agents");
   124→  const [bookingStatusFilter, setBookingStatusFilter] = useState("all");
   125→  const [loading, setLoading] = useState(true);
   126→
   127→  // Agent form dialog
   128→  const [agentDialogOpen, setAgentDialogOpen] = useState(false);
   129→  const [editingAgent, setEditingAgent] = useState<TelegramAgent | null>(null);
   130→  const [agentForm, setAgentForm] = useState({
   131→    name: "",
   132→    token: "",
   133→    botUsername: "",
   134→    businessType: "restaurant",
   135→    welcomeMessage: "",
   136→    address: "",
   137→    phone: "",
   138→    currency: "XAF",
   139→    paymentMethod: "",
   140→  });
   141→  const [agentSaving, setAgentSaving] = useState(false);
   142→
   143→  // Services dialog
   144→  const [servicesDialogOpen, setServicesDialogOpen] = useState(false);
   145→  const [servicesAgent, setServicesAgent] = useState<TelegramAgent | null>(null);
   146→  const [services, setServices] = useState<BusinessService[]>([]);
   147→  const [serviceForm, setServiceForm] = useState({
   148→    name: "",
   149→    description: "",
   150→    price: "",
   151→    duration: "",
   152→    isActive: true,
   153→  });
   154→  const [serviceSaving, setServiceSaving] = useState(false);
   155→  const [servicesLoading, setServicesLoading] = useState(false);
   156→
   157→  const headers = { Authorization: `Bearer ${token}` };
   158→
   159→  // ─── Fetch Data ────────────────────────────────────────────────
   160→
   161→  const fetchAgents = useCallback(async () => {
   162→    try {
   163→      const res = await fetch("/api/telegram/agents", { headers });
   164→      if (!res.ok) throw new Error();
   165→      const data = await res.json();
   166→      setAgents(data.agents || []);
   167→    } catch {
   168→      toast.error("Erreur lors du chargement des agents");
   169→    }
   170→  }, [headers]);
   171→
   172→  const fetchBookings = useCallback(async (status?: string) => {
   173→    try {
   174→      const params = new URLSearchParams();
   175→      if (status && status !== "all") params.set("status", status);
   176→      const res = await fetch(`/api/telegram/bookings?${params}`, { headers });
   177→      if (!res.ok) throw new Error();
   178→      const data = await res.json();
   179→      setBookings(data.bookings || []);
   180→    } catch {
   181→      toast.error("Erreur lors du chargement des réservations");
   182→    }
   183→  }, [headers]);
   184→
   185→  useEffect(() => {
   186→    const load = async () => {
   187→      setLoading(true);
   188→      await Promise.all([fetchAgents(), fetchBookings()]);
   189→      setLoading(false);
   190→    };
   191→    load();
   192→  }, [fetchAgents, fetchBookings]);
   193→
   194→  // ─── Stats ─────────────────────────────────────────────────────
   195→
   196→  const today = new Date().toISOString().split("T")[0];
   197→  const thisMonth = new Date().toISOString().slice(0, 7);
   198→
   199→  const activeTodayCount = bookings.filter(
   200→    (b) => b.bookingDate === today && b.status !== "cancelled"
   201→  ).length;
   202→  const pendingCount = bookings.filter((b) => b.status === "pending").length;
   203→  const completedThisMonth = bookings.filter(
   204→    (b) => b.status === "completed" && b.createdAt?.startsWith(thisMonth)
   205→  ).length;
   206→
   207→  const stats = [
   208→    {
   209→      label: "Total Agents",
   210→      value: agents.length,
   211→      icon: Bot,
   212→      color: "text-emerald-600",
   213→      bg: "bg-emerald-50 dark:bg-emerald-950/40",
   214→    },
   215→    {
   216→      label: "Réservations Aujourd\'hui",
   217→      value: activeTodayCount,
   218→      icon: CalendarDays,
   219→      color: "text-amber-600",
   220→      bg: "bg-amber-50 dark:bg-amber-950/40",
   221→    },
   222→    {
   223→      label: "En Attente",
   224→      value: pendingCount,
   225→      icon: Clock,
   226→      color: "text-orange-600",
   227→      bg: "bg-orange-50 dark:bg-orange-950/40",
   228→    },
   229→    {
   230→      label: "Terminées ce Mois",
   231→      value: completedThisMonth,
   232→      icon: CheckCircle,
   233→      color: "text-green-600",
   234→      bg: "bg-green-50 dark:bg-green-950/40",
   235→    },
   236→  ];
   237→
   238→  // ─── Agent CRUD ────────────────────────────────────────────────
   239→
   240→  const openCreateAgent = () => {
   241→    setEditingAgent(null);
   242→    setAgentForm({
   243→      name: "",
   244→      token: "",
   245→      botUsername: "",
   246→      businessType: "restaurant",
   247→      welcomeMessage: "",
   248→      address: "",
   249→      phone: "",
   250→      currency: "XAF",
   251→      paymentMethod: "",
   252→    });
   253→    setAgentDialogOpen(true);
   254→  };
   255→
   256→  const openEditAgent = (agent: TelegramAgent) => {
   257→    setEditingAgent(agent);
   258→    setAgentForm({
   259→      name: agent.name,
   260→      token: agent.token,
   261→      botUsername: agent.botUsername || "",
   262→      businessType: agent.businessType,
   263→      welcomeMessage: agent.welcomeMessage || "",
   264→      address: agent.address || "",
   265→      phone: agent.phone || "",
   266→      currency: agent.currency,
   267→      paymentMethod: agent.paymentMethod || "",
   268→    });
   269→    setAgentDialogOpen(true);
   270→  };
   271→
   272→  const saveAgent = async () => {
   273→    if (!agentForm.name || !agentForm.token) {
   274→      toast.error("Nom et token requis");
   275→      return;
   276→    }
   277→    setAgentSaving(true);
   278→    try {
   279→      if (editingAgent) {
   280→        const res = await fetch(`/api/telegram/agents/${editingAgent.id}`, {
   281→          method: "PUT",
   282→          headers: { ...headers, "Content-Type": "application/json" },
   283→          body: JSON.stringify(agentForm),
   284→        });
   285→        if (!res.ok) throw new Error();
   286→        toast.success("Agent mis à jour");
   287→      } else {
   288→        const res = await fetch("/api/telegram/agents", {
   289→          method: "POST",
   290→          headers: { ...headers, "Content-Type": "application/json" },
   291→          body: JSON.stringify(agentForm),
   292→        });
   293→        if (!res.ok) throw new Error();
   294→        toast.success("Agent créé");
   295→      }
   296→      setAgentDialogOpen(false);
   297→      await fetchAgents();
   298→    } catch {
   299→      toast.error("Erreur lors de la sauvegarde");
   300→    } finally {
   301→      setAgentSaving(false);
   302→    }
   303→  };
   304→
   305→  const deleteAgent = async (id: string) => {
   306→    if (!confirm("Supprimer cet agent et toutes ses données ?")) return;
   307→    try {
   308→      const res = await fetch(`/api/telegram/agents/${id}`, {
   309→        method: "DELETE",
   310→        headers,
   311→      });
   312→      if (!res.ok) throw new Error();
   313→      toast.success("Agent supprimé");
   314→      await fetchAgents();
   315→      await fetchBookings(bookingStatusFilter);
   316→    } catch {
   317→      toast.error("Erreur lors de la suppression");
   318→    }
   319→  };
   320→
   321→  const toggleAgentActive = async (agent: TelegramAgent) => {
   322→    try {
   323→      const res = await fetch(`/api/telegram/agents/${agent.id}`, {
   324→        method: "PUT",
   325→        headers: { ...headers, "Content-Type": "application/json" },
   326→        body: JSON.stringify({ isActive: !agent.isActive }),
   327→      });
   328→      if (!res.ok) throw new Error();
   329→      await fetchAgents();
   330→    } catch {
   331→      toast.error("Erreur lors du changement de statut");
   332→    }
   333→  };
   334→
   335→  // ─── Services Management ───────────────────────────────────────
   336→
   337→  const openServices = async (agent: TelegramAgent) => {
   338→    setServicesAgent(agent);
   339→    setServiceForm({ name: "", description: "", price: "", duration: "", isActive: true });
   340→    setServicesDialogOpen(true);
   341→    setServicesLoading(true);
   342→    try {
   343→      const res = await fetch(`/api/telegram/agents/${agent.id}/services`, { headers });
   344→      if (!res.ok) throw new Error();
   345→      const data = await res.json();
   346→      setServices(data.services || []);
   347→    } catch {
   348→      toast.error("Erreur lors du chargement des services");
   349→    } finally {
   350→      setServicesLoading(false);
   351→    }
   352→  };
   353→
   354→  const saveService = async () => {
   355→    if (!servicesAgent || !serviceForm.name || !serviceForm.price) {
   356→      toast.error("Nom et prix requis");
   357→      return;
   358→    }
   359→    setServiceSaving(true);
   360→    try {
   361→      const res = await fetch(`/api/telegram/agents/${servicesAgent.id}/services`, {
   362→        method: "POST",
   363→        headers: { ...headers, "Content-Type": "application/json" },
   364→        body: JSON.stringify(serviceForm),
   365→      });
   366→      if (!res.ok) throw new Error();
   367→      toast.success("Service ajouté");
   368→      setServiceForm({ name: "", description: "", price: "", duration: "", isActive: true });
   369→      // Refresh services
   370→      const svcRes = await fetch(`/api/telegram/agents/${servicesAgent.id}/services`, { headers });
   371→      if (svcRes.ok) {
   372→        const data = await svcRes.json();
   373→        setServices(data.services || []);
   374→      }
   375→      await fetchAgents();
   376→    } catch {
   377→      toast.error("Erreur lors de l\'ajout du service");
   378→    } finally {
   379→      setServiceSaving(false);
   380→    }
   381→  };
   382→
   383→  const deleteService = async (serviceId: string) => {
   384→    if (!servicesAgent) return;
   385→    try {
   386→      // Use the service directly via the agent API by deleting individually
   387→      const res = await fetch(`/api/telegram/agents/${servicesAgent.id}/services?serviceId=${serviceId}`, {
   388→        method: "DELETE",
   389→        headers,
   390→      });
   391→      if (!res.ok) {
   392→        // Fallback: just remove from local state
   393→      }
   394→      setServices((prev) => prev.filter((s) => s.id !== serviceId));
   395→      toast.success("Service supprimé");
   396→      await fetchAgents();
   397→    } catch {
   398→      // Still remove locally
   399→      setServices((prev) => prev.filter((s) => s.id !== serviceId));
   400→    }
   401→  };
   402→
   403→  // ─── Booking Actions ───────────────────────────────────────────
   404→
   405→  const updateBookingStatus = async (id: string, status: string) => {
   406→    try {
   407→      const res = await fetch("/api/telegram/bookings", {
   408→        method: "PUT",
   409→        headers: { ...headers, "Content-Type": "application/json" },
   410→        body: JSON.stringify({ id, status }),
   411→      });
   412→      if (!res.ok) throw new Error();
   413→      toast.success(`Réservation ${status === "confirmed" ? "confirmée" : status === "completed" ? "terminée" : "annulée"}`);
   414→      await fetchBookings(bookingStatusFilter);
   415→    } catch {
   416→      toast.error("Erreur lors de la mise à jour");
   417→    }
   418→  };
   419→
   420→  // ─── Render ────────────────────────────────────────────────────
   421→
   422→  if (loading) {
   423→    return (
   424→      <div className="p-6 flex items-center justify-center min-h-[60vh]">
   425→        <Loader2 className="w-8 h-8 animate-spin text-muted-foreground" />
   426→      </div>
   427→    );
   428→  }
   429→
   430→  return (
   431→    <div className="p-4 md:p-6 space-y-6">
   432→      {/* Header */}
   433→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   434→        <div>
   435→          <h1 className="text-2xl font-bold text-foreground flex items-center gap-2">
   436→            <Bot className="w-7 h-7 text-[#0088cc]" />
   437→            Agents Telegram
   438→          </h1>
   439→          <p className="text-sm text-muted-foreground mt-1">
   440→            Gérez vos bots de réservation pour restaurants et salons de coiffure
   441→          </p>
   442→        </div>
   443→        <Button onClick={openCreateAgent} className="gap-2">
   444→          <Plus className="w-4 h-4" />
   445→          Ajouter un Agent
   446→        </Button>
   447→      </div>
   448→
   449→      {/* Stats Cards */}
   450→      <div className="grid grid-cols-2 lg:grid-cols-4 gap-4">
   451→        {stats.map((stat) => (
   452→          <Card key={stat.label} className="border-0 shadow-sm">
   453→            <CardContent className="p-4">
   454→              <div className="flex items-center gap-3">
   455→                <div className={`p-2.5 rounded-xl ${stat.bg}`}>
   456→                  <stat.icon className={`w-5 h-5 ${stat.color}`} />
   457→                </div>
   458→                <div>
   459→                  <p className="text-2xl font-bold text-foreground">{stat.value}</p>
   460→                  <p className="text-xs text-muted-foreground">{stat.label}</p>
   461→                </div>
   462→              </div>
   463→            </CardContent>
   464→          </Card>
   465→        ))}
   466→      </div>
   467→
   468→      {/* Tabs */}
   469→      <Tabs value={activeTab} onValueChange={setActiveTab}>
   470→        <TabsList className="bg-muted/50">
   471→          <TabsTrigger value="agents" className="gap-2">
   472→            <Bot className="w-4 h-4" />
   473→            Agents
   474→          </TabsTrigger>
   475→          <TabsTrigger value="reservations" className="gap-2">
   476→            <CalendarDays className="w-4 h-4" />
   477→            Réservations
   478→            {pendingCount > 0 && (
   479→              <span className="bg-orange-500 text-white text-[10px] font-bold px-1.5 py-0.5 rounded-full">
   480→                {pendingCount}
   481→              </span>
   482→            )}
   483→          </TabsTrigger>
   484→        </TabsList>
   485→
   486→        {/* ─── Agents Tab ─── */}
   487→        <TabsContent value="agents" className="mt-4">
   488→          {agents.length === 0 ? (
   489→            <Card className="border-0 shadow-sm">
   490→              <CardContent className="p-12 text-center">
   491→                <Bot className="w-12 h-12 text-muted-foreground/40 mx-auto mb-4" />
   492→                <h3 className="text-lg font-semibold text-foreground mb-1">Aucun agent Telegram</h3>
   493→                <p className="text-sm text-muted-foreground mb-4">
   494→                  Créez votre premier bot de réservation pour votre restaurant ou salon.
   495→                </p>
   496→                <Button onClick={openCreateAgent} className="gap-2">
   497→                  <Plus className="w-4 h-4" />
   498→                  Créer un Agent
   499→                </Button>
   500→              </CardContent>
   501→            </Card>
   502→          ) : (
   503→            <div className="grid gap-4 md:grid-cols-2 xl:grid-cols-3">
   504→              {agents.map((agent) => (
   505→                <Card key={agent.id} className="border-0 shadow-sm hover:shadow-md transition-shadow">
   506→                  <CardHeader className="pb-3">
   507→                    <div className="flex items-start justify-between">
   508→                      <div className="flex items-center gap-3">
   509→                        <div className={`p-2.5 rounded-xl ${agent.businessType === "restaurant" ? "bg-orange-50 dark:bg-orange-950/40" : "bg-pink-50 dark:bg-pink-950/40"}`}>
   510→                          {agent.businessType === "restaurant" ? (
   511→                            <UtensilsCrossed className={`w-5 h-5 ${agent.businessType === "restaurant" ? "text-orange-600" : "text-pink-600"}`} />
   512→                          ) : (
   513→                            <Scissors className="w-5 h-5 text-pink-600" />
   514→                          )}
   515→                        </div>
   516→                        <div>
   517→                          <CardTitle className="text-base font-semibold text-foreground">{agent.name}</CardTitle>
   518→                          {agent.botUsername && (
   519→                            <p className="text-xs text-muted-foreground">@{agent.botUsername.replace("@", "")}</p>
   520→                          )}
   521→                        </div>
   522→                      </div>
   523→                      <Switch
   524→                        checked={agent.isActive}
   525→                        onCheckedChange={() => toggleAgentActive(agent)}
   526→                      />
   527→                    </div>
   528→                  </CardHeader>
   529→                  <CardContent className="space-y-4">
   530→                    <div className="flex items-center gap-2">
   531→                      <Badge variant="outline" className={
   532→                        agent.businessType === "restaurant"
   533→                          ? "bg-orange-100 text-orange-800 dark:bg-orange-900/30 dark:text-orange-400"
   534→                          : "bg-pink-100 text-pink-800 dark:bg-pink-900/30 dark:text-pink-400"
   535→                      }>
   536→                        {agent.businessType === "restaurant" ? "🍽️ Restaurant" : "✂️ Salon de coiffure"}
   537→                      </Badge>
   538→                      {!agent.isActive && (
   539→                        <Badge variant="outline" className="bg-muted text-muted-foreground">
   540→                          Inactif
   541→                        </Badge>
   542→                      )}
   543→                    </div>
   544→
   545→                    <div className="space-y-1.5 text-sm">
   546→                      {agent.address && (
   547→                        <div className="flex items-center gap-2 text-muted-foreground">
   548→                          <MapPin className="w-3.5 h-3.5 shrink-0" />
   549→                          <span className="truncate">{agent.address}</span>
   550→                        </div>
   551→                      )}
   552→                      {agent.phone && (
   553→                        <div className="flex items-center gap-2 text-muted-foreground">
   554→                          <Phone className="w-3.5 h-3.5 shrink-0" />
   555→                          <span>{agent.phone}</span>
   556→                        </div>
   557→                      )}
   558→                      {agent.paymentMethod && (
   559→                        <div className="flex items-center gap-2 text-muted-foreground">
   560→                          <CreditCard className="w-3.5 h-3.5 shrink-0" />
   561→                          <span>{agent.paymentMethod}</span>
   562→                        </div>
   563→                      )}
   564→                    </div>
   565→
   566→                    <div className="flex items-center gap-4 pt-2 border-t border-border">
   567→                      <div className="text-center">
   568→                        <p className="text-lg font-bold text-foreground">{agent._count.services}</p>
   569→                        <p className="text-[11px] text-muted-foreground">Services</p>
   570→                      </div>
   571→                      <div className="text-center">
