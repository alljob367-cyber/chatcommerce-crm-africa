   192→function authHeaders() {
   193→  const token =
   194→    typeof window !== "undefined"
   195→      ? localStorage.getItem("cc_token")
   196→      : null;
   197→  return {
   198→    "Content-Type": "application/json",
   199→    Authorization: token ? `Bearer ${token}` : "",
   200→  };
   201→}
