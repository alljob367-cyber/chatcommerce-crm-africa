   127→/* ─── ANIMATED COUNTER HOOK ─────────────────────────── */
   128→
   129→function useCounter(end: number, duration: number = 2000) {
   130→  const [count, setCount] = useState(0);
   131→  const ref = useRef<HTMLDivElement>(null);
   132→  const started = useRef(false);
   133→
   134→  useEffect(() => {
   135→    const observer = new IntersectionObserver(
   136→      ([entry]) => {
   137→        if (entry.isIntersecting && !started.current) {
   138→          started.current = true;
   139→          const start = Date.now();
   140→          const animate = () => {
   141→            const elapsed = Date.now() - start;
   142→            const progress = Math.min(elapsed / duration, 1);
   143→            setCount(Math.floor(progress * end));
   144→            if (progress < 1) requestAnimationFrame(animate);
   145→          };
   146→          animate();
   147→        }
   148→      },
   149→      { threshold: 0.3 }
   150→    );
   151→    if (ref.current) observer.observe(ref.current);
   152→    return () => observer.disconnect();
   153→  }, [end, duration]);
   154→
   155→  return { count, ref };
   156→}
