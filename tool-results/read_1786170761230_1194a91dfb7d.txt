   157→   157→function generateId(): string {
   158→   158→  const chars = "abcdefghijklmnopqrstuvwxyz0123456789";
   159→   159→  let id = "";
   160→   160→  const bytes = new Uint8Array(25);

   168→   168→const DAY_KEYS: Record<string, string> = {
   169→   169→  "1": "lun", "2": "mar", "3": "mer", "4": "jeu",
   170→   170→  "5": "ven", "6": "sam", "0": "dim",

   174→   174→  lun: "Lundi", mar: "Mardi", mer: "Mercredi", jeu: "Jeudi",
   175→   175→  ven: "Vendredi", sam: "Samedi", dim: "Dimanche",

   178→   178→function isOpenNow(agent: TelegramAgent): { open: boolean; hours?: string } {
   179→   179→  if (!agent.openHours) return { open: true }; // No hours = always open
   180→   180→  try {
   181→   181→    const hours = JSON.parse(agent.openHours) as Record<string, string>;
   182→   182→    const now = new Date();
   183→   183→    const dayKey = DAY_KEYS[String(now.getDay())] || "lun";
   184→   184→    const todayHours = hours[dayKey];
   185→   185→    if (!todayHours) return { open: false };
   186→   186→    const [open, close] = todayHours.split("-").map((t: string) => t.trim());
   187→   187→    const currentMinutes = now.getHours() * 60 + now.getMinutes();
   188→   188→    const [oh, om] = open.split(":").map(Number);
   189→   189→    const [ch, cm] = close.split(":").map(Number);
   190→   190→    return {
   191→   191→      open: currentMinutes >= oh * 60 + om && currentMinutes <= ch * 60 + cm,
   192→   192→      hours: todayHours,
   193→   193→    };
   194→   194→  } catch {
   195→   195→    return { open: true };
   196→   196→  }
   197→   197→}
   198→   198→
   199→   199→function formatOpenHours(agent: TelegramAgent): string {
   200→   200→  if (!agent.openHours) return "Non défini";
   201→   201→  try {
   202→   202→    const hours = JSON.parse(agent.openHours) as Record<string, string>;
   203→   203→    const lines = Object.entries(DAY_NAMES)
   204→   204→      .map(([key, name]) => {
   205→   205→        const h = hours[key];
   206→   206→        return h ? `  <b>${name}</b>: ${h}` : `  <b>${name}</b>: Fermé`;
   207→   207→      });
   208→   208→    return lines.join("\n");
   209→   209→  } catch {
   210→   210→    return "Erreur de format";
   211→   211→  }
   212→   212→}

   654→   654→function parseDate(input: string): string | null {
   655→   655→  const match = input.trim().match(/^(\d{1,2})[\/.-](\d{1,2})[\/.-](\d{4})$/);
   656→   656→  if (!match) return null;
   657→   657→  const [, d, m, y] = match;
   658→   658→  const day = parseInt(d);
   659→   659→  const month = parseInt(m);
   660→   660→  if (day < 1 || day > 31 || month < 1 || month > 12) return null;
   661→   661→  return `${y}-${String(month).padStart(2, "0")}-${String(day).padStart(2, "0")}`;
   662→   662→}
   663→   663→
   664→   664→function parseTime(input: string): string | null {
   665→   665→  const match = input.trim().match(/^(\d{1,2}):(\d{2})$/);
   666→   666→  if (!match) return null;
   667→   667→  const [, h, m] = match;
   668→   668→  const hour = parseInt(h);
   669→   669→  const min = parseInt(m);
   670→   670→  if (hour < 0 || hour > 23 || min < 0 || min > 59) return null;
   671→   671→  return `${String(hour).padStart(2, "0")}:${String(min).padStart(2, "0")}`;
   672→   672→}

   676→   676→async function processUpdate(token: string, agent: TelegramAgent, update: TelegramUpdate) {
   677→   677→  // Handle callback queries (button clicks)
   678→   678→  if (update.callback_query) {
   679→   679→    const cb = update.callback_query;
   680→   680→    const chatId = cb.from.id;
   681→   681→    const data = cb.data;
   682→   682→
   683→   683→    // Acknowledge callback
   684→   684→    await answerCallbackQuery(token, cb.id);
   685→   685→
   686→   686→    switch (data) {
   687→   687→      case "start":
   688→   688→        sessions.delete(chatId);
   689→   689→        await handleStart(token, agent, chatId, `${cb.from.first_name} ${cb.from.last_name || ""}`.trim());
   690→   690→        break;
   691→   691→      case "menu":
   692→   692→        await handleMenu(token, agent, chatId);
   693→   693→        break;
   694→   694→      case "cart":
   695→   695→        await handleViewCart(token, agent, chatId);
   696→   696→        break;
   697→   697→      case "clear_cart":
   698→   698→        await handleClearCart(token, agent, chatId);
   699→   699→        break;
   700→   700→      case "checkout":
   701→   701→        await handleCheckout(token, agent, chatId);
   702→   702→        break;
   703→   703→      case "contact":
   704→   704→        await handleContact(token, agent, chatId);
   705→   705→        break;
   706→   706→      case "horaire":
   707→   707→        await handleHoraires(token, agent, chatId);
   708→   708→        break;
   709→   709→      case "confirm":
   710→   710→        await handleConfirm(token, agent, chatId, `${cb.from.first_name} ${cb.from.last_name || ""}`.trim());
   711→   711→        break;
   712→   712→      case "cancel":
   713→   713→        await handleCancel(token, chatId);
   714→   714→        break;
   715→   715→      default:
   716→   716→        if (data.startsWith("add_")) {
   717→   717→          const serviceId = data.slice(4);
   718→   718→          await handleAddToCart(token, agent, chatId, serviceId);
   719→   719→        }
   720→   720→    }
   721→   721→    return;
   722→   722→  }
   723→   723→
   724→   724→  // Handle text messages
   725→   725→  if (update.message) {
   726→   726→    const msg = update.message;
   727→   727→    const chatId = msg.chat.id;
   728→   728→    const name = getCustomerName(msg);
   729→   729→
   730→   730→    // Handle contact sharing
   731→   731→    if (msg.contact) {
   732→   732→      const session = sessions.get(chatId);
   733→   733→      if (session && session.state === "awaiting_phone") {
   734→   734→        await handlePhoneReceived(token, agent, chatId, msg.contact.phone_number, msg.contact.first_name || name);
   735→   735→        return;
   736→   736→      }
   737→   737→    }
   738→   738→
   739→   739→    if (!msg.text) return;
   740→   740→    const text = msg.text.trim();
   741→   741→    const session = sessions.get(chatId);
   742→   742→
   743→   743→    // Commands
   744→   744→    if (text === "/start") {
   745→   745→      sessions.delete(chatId);
   746→   746→      await handleStart(token, agent, chatId, name);
   747→   747→      return;
   748→   748→    }
   749→   749→    if (text === "/menu" || text === "/services") {
   750→   750→      await handleMenu(token, agent, chatId);
   751→   751→      return;
   752→   752→    }
   753→   753→    if (text === "/horaire" || text === "/hours") {
   754→   754→      await handleHoraires(token, agent, chatId);
   755→   755→      return;
   756→   756→    }
   757→   757→    if (text === "/contact") {
   758→   758→      await handleContact(token, agent, chatId);
   759→   759→      return;
   760→   760→    }
   761→   761→    if (text === "/aide" || text === "/help") {
   762→   762→      await handleAide(token, agent, chatId);
   763→   763→      return;
   764→   764→    }
   765→   765→
   766→   766→    // Session-based input
   767→   767→    if (session && session.agentId === agent.id) {
   768→   768→      switch (session.state) {
   769→   769→        case "awaiting_date":
   770→   770→          await handleDateReceived(token, agent, chatId, text);
   771→   771→          return;
   772→   772→        case "awaiting_time":
   773→   773→          await handleTimeReceived(token, agent, chatId, text);
   774→   774→          return;
   775→   775→        case "awaiting_phone":
   776→   776→          await handlePhoneReceived(token, agent, chatId, text, name);
   777→   777→          return;
   778→   778→      }
   779→   779→    }
   780→   780→
   781→   781→    // Default: show welcome
   782→   782→    await handleStart(token, agent, chatId, name);
   783→   783→  }
   784→   784→}

   788→   788→async function pollAgent(token: string, agent: TelegramAgent) {
   789→   789→  const offset = offsets.get(token) || 0;
   790→   790→  const result = await telegramApi(token, "getUpdates", {
   791→   791→    offset: offset + 1,
   792→   792→    timeout: 1,
   793→   793→    allowed_updates: ["message", "callback_query"],
   794→   794→  });
   795→   795→
   796→   796→  if (!result || !result.ok) {
   797→   797→    if (result && result.error_code === 401) {
   798→   798→      console.error(`[Bot] Invalid token for agent ${agent.name}, deactivating...`);
   799→   799→      db.query(`UPDATE TelegramAgent SET isActive = 0 WHERE id = ?`).run(agent.id);
   800→   800→      agentsMap.delete(token);
   801→   801→    }
   802→   802→    return;
   803→   803→  }
   804→   804→
   805→   805→  const updates: TelegramUpdate[] = result.result || [];
   806→   806→  for (const update of updates) {
   807→   807→    offsets.set(token, update.update_id);
   808→   808→    try {
   809→   809→      await processUpdate(token, agent, update);
   810→   810→    } catch (err) {
   811→   811→      console.error(`[Bot] Error processing update ${update.update_id} for ${agent.name}:`, err);
   812→   812→    }
   813→   813→  }
   814→   814→}
