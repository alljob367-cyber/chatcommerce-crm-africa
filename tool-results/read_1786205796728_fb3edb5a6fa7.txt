   332→   332→export async function POST(request: Request) {
   333→   333→  try {
   334→   334→    const session = await auth(request);
   335→   335→    if (!session) return NextResponse.json({ error: "Non autorisé" }, { status: 401 });
   336→   336→
   337→   337→    const companyId = session.companyId;
   338→   338→    const body = await request.json();
   339→   339→    const agentType = body.agentType; // optional: create specific type
   340→   340→
   341→   341→    // Check existing agents
   342→   342→    const existingAgents = await db.telegramAgent.findMany({ where: { companyId } });
   343→   343→    const existingTypes = existingAgents.map((a) => a.businessType);
   344→   344→
   345→   345→    // Determine which agents to create
   346→   346→    const templatesToCreate = agentType
   347→   347→      ? AGENT_TEMPLATES.filter((t) => t.businessType === agentType && !existingTypes.includes(agentType))
   348→   348→      : AGENT_TEMPLATES.filter((t) => !existingTypes.includes(t.businessType));
   349→   349→
   350→   350→    if (templatesToCreate.length === 0) {
   351→   351→      return NextResponse.json({
   352→   352→        success: true,
   353→   353→        message: agentType ? `Agent ${agentType} déjà configuré` : "Tous les agents sont déjà configurés",
   354→   354→        agents: existingAgents.length,
   355→   355→      });
   356→   356→    }
   357→   357→
   358→   358→    const created: { type: string; name: string; services: number }[] = [];
   359→   359→
   360→   360→    for (const template of templatesToCreate) {
   361→   361→      const agent = await db.telegramAgent.create({
   362→   362→        data: {
   363→   363→          companyId,
   364→   364→          name: template.name,
   365→   365→          token: template.token,
   366→   366→          botUsername: template.botUsername,
   367→   367→          businessType: template.businessType,
   368→   368→          isActive: false,
   369→   369→          welcomeMessage: template.welcomeMessage,
   370→   370→          address: template.address,
   371→   371→          phone: template.phone,
   372→   372→          openHours: template.openHours,
   373→   373→          currency: template.currency,
   374→   374→          paymentMethod: template.paymentMethod,
   375→   375→        },
   376→   376→      });
   377→   377→
   378→   378→      const services = ALL_SERVICES[template.businessType] || [];
   379→   379→      for (const svc of services) {
   380→   380→        await db.businessService.create({
   381→   381→          data: {
   382→   382→            agentId: agent.id,
   383→   383→            name: svc.name,
   384→   384→            description: svc.description,
   385→   385→            price: svc.price,
   386→   386→            duration: svc.duration,
   387→   387→            sortOrder: svc.sortOrder,
   388→   388→            isActive: true,
   389→   389→          },
   390→   390→        });
   391→   391→      }
   392→   392→
   393→   393→      created.push({ type: template.businessType, name: template.name, services: services.length });
   394→   394→    }
   395→   395→
   396→   396→    return NextResponse.json({
   397→   397→      success: true,
   398→   398→      message: `${created.length} agent(s) créé(s) avec succès !`,
   399→   399→      agents: created,
   400→   400→      totalAgents: existingAgents.length + created.length,
   401→   401→    });
   402→   402→  } catch (error: unknown) {
   403→   403→    const { error: msg, status } = handleError(error);
   404→   404→    return NextResponse.json({ error: msg }, { status });
   405→   405→  }
