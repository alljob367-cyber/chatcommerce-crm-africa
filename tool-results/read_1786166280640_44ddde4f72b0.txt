   157→function generateId(): string {
   158→  const chars = "abcdefghijklmnopqrstuvwxyz0123456789";
   159→  let id = "";
   160→  const bytes = new Uint8Array(25);
   161→  crypto.getRandomValues(bytes);
   162→  for (let i = 0; i < 25; i++) id += chars[bytes[i] % chars.length];
   163→  return id;
   164→}

   168→const DAY_KEYS: Record<string, string> = {
   169→  "1": "lun", "2": "mar", "3": "mer", "4": "jeu",
   170→  "5": "ven", "6": "sam", "0": "dim",
   171→};
   172→
   173→const DAY_NAMES: Record<string, string> = {
   174→  lun: "Lundi", mar: "Mardi", mer: "Mercredi", jeu: "Jeudi",
   175→  ven: "Vendredi", sam: "Samedi", dim: "Dimanche",
   176→};

   178→function isOpenNow(agent: TelegramAgent): { open: boolean; hours?: string } {
   179→  if (!agent.openHours) return { open: true }; // No hours = always open
   180→  try {
   181→    const hours = JSON.parse(agent.openHours) as Record<string, string>;
   182→    const now = new Date();
   183→    const dayKey = DAY_KEYS[String(now.getDay())] || "lun";
   184→    const todayHours = hours[dayKey];
   185→    if (!todayHours) return { open: false };
   186→    const [open, close] = todayHours.split("-").map((t: string) => t.trim());
   187→    const currentMinutes = now.getHours() * 60 + now.getMinutes();
   188→    const [oh, om] = open.split(":").map(Number);
   189→    const [ch, cm] = close.split(":").map(Number);
   190→    return {
   191→      open: currentMinutes >= oh * 60 + om && currentMinutes <= ch * 60 + cm,
   192→      hours: todayHours,
   193→    };
   194→  } catch {
   195→    return { open: true };
   196→  }
   197→}

   199→function formatOpenHours(agent: TelegramAgent): string {
   200→  if (!agent.openHours) return "Non défini";
   201→  try {
   202→    const hours = JSON.parse(agent.openHours) as Record<string, string>;
   203→    const lines = Object.entries(DAY_NAMES)
   204→      .map(([key, name]) => {
   205→        const h = hours[key];
   206→        return h ? `  <b>${name}</b>: ${h}` : `  <b>${name}</b>: Fermé`;
   207→      });
   208→    return lines.join("\n");
   209→  } catch {
   210→    return "Erreur de format";
   211→  }
   212→}

   654→function parseDate(input: string): string | null {
   655→  const match = input.trim().match(/^(\d{1,2})[\/.-](\d{1,2})[\/.-](\d{4})$/);
   656→  if (!match) return null;
   657→  const [, d, m, y] = match;
   658→  const day = parseInt(d);
   659→  const month = parseInt(m);
   660→  if (day < 1 || day > 31 || month < 1 || month > 12) return null;
   661→  return `${y}-${String(month).padStart(2, "0")}-${String(day).padStart(2, "0")}`;
   662→}
   663→
   664→function parseTime(input: string): string | null {
   665→  const match = input.trim().match(/^(\d{1,2}):(\d{2})$/);
   666→  if (!match) return null;
   667→  const [, h, m] = match;
   668→  const hour = parseInt(h);
   669→  const min = parseInt(m);
   670→  if (hour < 0 || hour > 23 || min < 0 || min > 59) return null;
   671→  return `${String(hour).padStart(2, "0")}:${String(min).padStart(2, "0")}`;
   672→}

   676→async function processUpdate(token: string, agent: TelegramAgent, update: TelegramUpdate) {
   677→  // Handle callback queries (button clicks)
   678→  if (update.callback_query) {
   679→    const cb = update.callback_query;
   680→    const chatId = cb.from.id;
   681→    const data = cb.data;
   682→
   683→    // Acknowledge callback
   684→    await answerCallbackQuery(token, cb.id);
   685→
   686→    switch (data) {
   687→      case "start":
   688→        sessions.delete(chatId);
   689→        await handleStart(token, agent, chatId, `${cb.from.first_name} ${cb.from.last_name || ""}`.trim());
   690→        break;
   691→      case "menu":
   692→        await handleMenu(token, agent, chatId);
   693→        break;
   694→      case "cart":
   695→        await handleViewCart(token, agent, chatId);
   696→        break;
   697→      case "clear_cart":
   698→        await handleClearCart(token, agent, chatId);
   699→        break;
   700→      case "checkout":
   701→        await handleCheckout(token, agent, chatId);
   702→        break;
   703→      case "contact":
   704→        await handleContact(token, agent, chatId);
   705→        break;
   706→      case "horaire":
   707→        await handleHoraires(token, agent, chatId);
   708→        break;
   709→      case "confirm":
   710→        await handleConfirm(token, agent, chatId, `${cb.from.first_name} ${cb.from.last_name || ""}`.trim());
   711→        break;
   712→      case "cancel":
   713→        await handleCancel(token, chatId);
   714→        break;
   715→      default:
   716→        if (data.startsWith("add_")) {
   717→          const serviceId = data.slice(4);
   718→          await handleAddToCart(token, agent, chatId, serviceId);
   719→        }
   720→    }
   721→    return;
   722→  }
   723→
   724→  // Handle text messages
   725→  if (update.message) {
   726→    const msg = update.message;
   727→    const chatId = msg.chat.id;
   728→    const name = getCustomerName(msg);
   729→
   730→    // Handle contact sharing
   731→    if (msg.contact) {
   732→      const session = sessions.get(chatId);
   733→      if (session && session.state === "awaiting_phone") {
   734→        await handlePhoneReceived(token, agent, chatId, msg.contact.phone_number, msg.contact.first_name || name);
   735→        return;
   736→      }
   737→    }
   738→
   739→    if (!msg.text) return;
   740→    const text = msg.text.trim();
   741→    const session = sessions.get(chatId);
   742→
   743→    // Commands
   744→    if (text === "/start") {
   745→      sessions.delete(chatId);
   746→      await handleStart(token, agent, chatId, name);
   747→      return;
   748→    }
   749→    if (text === "/menu" || text === "/services") {
   750→      await handleMenu(token, agent, chatId);
   751→      return;
   752→    }
   753→    if (text === "/horaire" || text === "/hours") {
   754→      await handleHoraires(token, agent, chatId);
   755→      return;
   756→    }
   757→    if (text === "/contact") {
   758→      await handleContact(token, agent, chatId);
   759→      return;
   760→    }
   761→    if (text === "/aide" || text === "/help") {
   762→      await handleAide(token, agent, chatId);
   763→      return;
   764→    }
   765→
   766→    // Session-based input
   767→    if (session && session.agentId === agent.id) {
   768→      switch (session.state) {
   769→        case "awaiting_date":
   770→          await handleDateReceived(token, agent, chatId, text);
   771→          return;
   772→        case "awaiting_time":
   773→          await handleTimeReceived(token, agent, chatId, text);
   774→          return;
   775→        case "awaiting_phone":
   776→          await handlePhoneReceived(token, agent, chatId, text, name);
   777→          return;
   778→      }
   779→    }
   780→
   781→    // Default: show welcome
   782→    await handleStart(token, agent, chatId, name);
   783→  }
   784→}

   788→async function pollAgent(token: string, agent: TelegramAgent) {
   789→  const offset = offsets.get(token) || 0;
   790→  const result = await telegramApi(token, "getUpdates", {
   791→    offset: offset + 1,
   792→    timeout: 1,
   793→    allowed_updates: ["message", "callback_query"],
   794→  });
   795→
   796→  if (!result || !result.ok) {
   797→    if (result && result.error_code === 401) {
   798→      console.error(`[Bot] Invalid token for agent ${agent.name}, deactivating...`);
   799→      db.query(`UPDATE TelegramAgent SET isActive = 0 WHERE id = ?`).run(agent.id);
   800→      agentsMap.delete(token);
   801→    }
   802→    return;
   803→  }
   804→
   805→  const updates: TelegramUpdate[] = result.result || [];
   806→  for (const update of updates) {
   807→    offsets.set(token, update.update_id);
   808→    try {
   809→      await processUpdate(token, agent, update);
   810→    } catch (err) {
   811→      console.error(`[Bot] Error processing update ${update.update_id} for ${agent.name}:`, err);
   812→    }
   813→  }
   814→}

   816→async function pollAll() {
   817→  for (const [token, agent] of agentsMap) {
   818→    await pollAgent(token, agent).catch((err) => {
   819→      console.error(`[Bot] Poll error for ${agent.name}:`, err);
   820→    });
   821→  }
   822→}

   826→function refreshAgents() {
   827→  const fresh = loadActiveAgents();
   828→  const newMap = new Map<string, TelegramAgent>();
   829→
   830→  for (const agent of fresh) {

   832→      console.log(`[Bot] New agent registered: ${agent.name} (${agent.businessType})`);
   833→    }
   834→    newMap.set(agent.token, agent);
   835→  }
   836→
   837→  for (const [token, agent] of agentsMap) {
   838→    if (!newMap.has(token)) {
   839→      console.log(`[Bot] Agent removed: ${agent.name}`);
   840→      offsets.delete(token);
   841→    }
   842→  }
   843→
   844→  agentsMap.clear();
   845→  for (const [k, v] of newMap) agentsMap.set(k, v);
   846→}
